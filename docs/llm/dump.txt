# zen-velixir-run

Minimal .NET 10 Blazor Server app on the [velixir](https://zen.velixir.run/) free tier (0.25 vCPU, 256 MB).

- Live message board: anyone posts up to 500 chars, everyone connected sees it instantly. In memory only, newest 50 kept, 2 s per-visitor cooldown.
- Header shows server time, ticking every second from one shared server timer.
- Beacons (`/beacons`): public, unauthenticated capture API with a live page and usage docs. See below.
- Packages: Dapper and Npgsql only. No CSS or JS frameworks. Evergreen browsers only.
- Component styles in `*.razor.css`; globals and color tokens (`light-dark()`) in `wwwroot/app.css`.
- Binds to `PORT` when set; honors `X-Forwarded-For`/`X-Forwarded-Proto` from the edge.
- Memory cap: workstation non-concurrent GC, `ConserveMemory=7`, no tiered PGO, invariant globalization, short disconnected-circuit retention, 2 unacked render batches per circuit.
- Data protection keys are in-memory; a restart invalidates antiforgery cookies, so that log category is off. Restarts show as `Application started` in logs.
- Messages truncate on grapheme boundaries.
- Every page sends `Content-Security-Policy-Report-Only`; reports land in beacon key `zen-csp`.
- Push to `main` builds then deploys via `.github/workflows/velixir.yml` (secret `VELIXIR_API_KEY`). PRs build only.

## Beacons

| Method | Path | Result |
|---|---|---|
| POST | `/api/beacons/{key}` | 201, UUIDv7 id, `Location` |
| GET | `/api/entries/{id}` | Full record or 404; 503 + `Retry-After` if DB down and not in memory |
| GET | `/api/beacons/{key}?referer=` | Total, bytes, first/last, top 25 referers, hourly 24 h |
| GET | `/api/beacons/{key}/entries?referer=&before=&limit=` | Newest first, max 100, `next` cursor |
| GET | `/api/beacons?limit=` | Busiest keys |
| GET | `/api/status` | Storage health |

- Key: `^[A-Za-z0-9][A-Za-z0-9._~:-]{0,63}$`. Body ≤ 32 KiB, any type; invalid UTF-8 or NUL stored as base64.
- Records method, path, query, referer (`Referer` else `Origin`), content type, user agent, headers, body. IP truncated to /24 or /48. `Cookie`, `Authorization` and forwarding headers redacted.
- Open CORS. Per-IP token buckets: 30 writes/min, 120 reads/min.
- Writes go to a bounded memory buffer (4 MiB / 2000 entries) and a background writer batches them into PostgreSQL. DB down: reads serve memory with `degraded: true`, writer retries with backoff (2–60 s), oldest unsaved entries drop if the buffer fills.
- Janitor every 2 min: deletes entries older than 30 days, trims oldest to 80% when stored bytes exceed 48 MiB, halves the budget and vacuums if `pg_database_size` exceeds 160 MiB and is still growing.
- Pool max 2 connections (server allows 5). Schema is created on first use.
- No PostgreSQL settings: memory only.

Settings in `appsettings.json` under `Beacons`. Connection from env vars `postgreshost`, `postgresport`, `postgresdatabase`, `postgresuser`, `postgrespassword`, or `postgresconnectionstring` (`postgresql://user:pass@host:port/db`) as fallback.

using Microsoft.Extensions.Options;

namespace Zen.Blazor.Beacons;

public readonly record struct BufferedBeacon(BeaconEntry Entry, bool Persisted);

public readonly record struct BufferStats(int Entries, long Bytes, int Unsaved, long Dropped);

public sealed class BeaconBuffer(IOptions<BeaconOptions> options, ILogger<BeaconBuffer> logger)
{
    private readonly BeaconOptions _options = options.Value;
    private readonly Lock _gate = new();
    private readonly LinkedList<Slot> _order = new();
    private readonly Dictionary<Guid, LinkedListNode<Slot>> _index = [];
    private readonly SemaphoreSlim _signal = new(0, 1);
    private long _bytes;
    private int _unsaved;
    private long _dropped;

    public event Action? Changed;

    public void Add(BeaconEntry entry)
    {
        lock (_gate)
        {
            _index[entry.Id] = _order.AddLast(new Slot(entry));
            _bytes += entry.Size;
            _unsaved++;

            while (_order.First is { } oldest && oldest != _order.Last
                   && (_bytes > _options.MemoryBytes || _order.Count > _options.MemoryEntries))
            {
                Remove(oldest);
                if (!oldest.Value.Persisted)
                {
                    _unsaved--;
                    _dropped++;
                }
            }
        }

        Signal();
        Notify();
    }

    public bool TryGet(Guid id, out BeaconEntry entry)
    {
        lock (_gate)
        {
            if (_index.TryGetValue(id, out var node))
            {
                entry = node.Value.Entry;
                return true;
            }
        }

        entry = null!;
        return false;
    }

    public IReadOnlyList<BeaconEntry> TakeUnsaved(int max)
    {
        var batch = new List<BeaconEntry>(Math.Min(max, 64));
        lock (_gate)
        {
            for (var node = _order.First; node is not null && batch.Count < max; node = node.Next)
            {
                if (!node.Value.Persisted)
                {
                    batch.Add(node.Value.Entry);
                }
            }
        }

        return batch;
    }

    public void MarkPersisted(IEnumerable<BeaconEntry> entries)
    {
        lock (_gate)
        {
            foreach (var entry in entries)
            {
                if (_index.TryGetValue(entry.Id, out var node) && !node.Value.Persisted)
                {
                    node.Value.Persisted = true;
                    _unsaved--;
                }
            }
        }
    }

    public IReadOnlyList<BufferedBeacon> Snapshot()
    {
        lock (_gate)
        {
            var items = new BufferedBeacon[_order.Count];
            var i = 0;
            foreach (var slot in _order)
            {
                items[i++] = new BufferedBeacon(slot.Entry, slot.Persisted);
            }

            return items;
        }
    }

    public BufferStats Stats
    {
        get
        {
            lock (_gate)
            {
                return new BufferStats(_order.Count, _bytes, _unsaved, _dropped);
            }
        }
    }

    public async Task WaitForWorkAsync(TimeSpan timeout, CancellationToken cancellationToken)
    {
        try
        {
            await _signal.WaitAsync(timeout, cancellationToken);
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
        }
    }

    private void Remove(LinkedListNode<Slot> node)
    {
        _order.Remove(node);
        _index.Remove(node.Value.Entry.Id);
        _bytes -= node.Value.Entry.Size;
    }

    private void Signal()
    {
        try
        {
            _signal.Release();
        }
        catch (SemaphoreFullException)
        {
        }
    }

    private void Notify()
    {
        if (Changed is not { } handlers)
        {
            return;
        }

        foreach (var handler in handlers.GetInvocationList().Cast<Action>())
        {
            try
            {
                handler();
            }
            catch (Exception ex)
            {
                logger.LogWarning(ex, "Beacon buffer subscriber failed");
            }
        }
    }

    private sealed class Slot(BeaconEntry entry)
    {
        public BeaconEntry Entry { get; } = entry;
        public bool Persisted { get; set; }
    }
}

using System.Buffers;
using System.Net;
using System.Net.Sockets;
using System.Text;
using System.Text.Unicode;
using Microsoft.Extensions.Options;

namespace Zen.Blazor.Beacons;

public sealed class BeaconCapture(IOptions<BeaconOptions> options, TimeProvider time)
{
    public const string Utf8Encoding = "utf-8";
    public const string Base64Encoding = "base64";
    public const string Redacted = "[redacted]";

    private const int MaxPathLength = 512;
    private const int MaxQueryLength = 2048;
    private const int MaxRefererLength = 512;
    private const int MaxUserAgentLength = 512;
    private const int MaxContentTypeLength = 256;
    private const int RowOverhead = 256;

    private static readonly HashSet<string> RedactedHeaders = new(StringComparer.OrdinalIgnoreCase)
    {
        "authorization",
        "proxy-authorization",
        "cookie",
        "x-forwarded-for",
        "x-real-ip",
        "x-client-ip",
        "x-original-for",
        "forwarded",
        "true-client-ip",
        "cf-connecting-ip",
        "fastly-client-ip",
        "x-cluster-client-ip",
    };

    private readonly BeaconOptions _options = options.Value;

    public int MaxBodyBytes => _options.MaxBodyBytes;

    public async Task<BeaconEntry?> FromRequestAsync(HttpContext context, string key, CancellationToken cancellationToken)
    {
        var request = context.Request;
        var limit = _options.MaxBodyBytes;
        if (request.ContentLength > limit)
        {
            return null;
        }

        var buffer = ArrayPool<byte>.Shared.Rent(limit + 1);
        try
        {
            var total = 0;
            while (total <= limit)
            {
                var read = await request.Body.ReadAsync(buffer.AsMemory(total, limit + 1 - total), cancellationToken);
                if (read == 0)
                {
                    break;
                }

                total += read;
            }

            if (total > limit)
            {
                return null;
            }

            var headers = CaptureHeaders(request.Headers);
            return Build(
                key,
                request.Method,
                request.Path.Value ?? "/",
                request.QueryString.Value ?? "",
                headers.GetValueOrDefault("referer") ?? headers.GetValueOrDefault("origin"),
                request.ContentType,
                Anonymize(context.Connection.RemoteIpAddress),
                headers.GetValueOrDefault("user-agent"),
                headers,
                buffer.AsSpan(0, total));
        }
        finally
        {
            ArrayPool<byte>.Shared.Return(buffer);
        }
    }

    public BeaconEntry FromPage(string key, string pageUrl, string body)
    {
        var headers = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase)
        {
            ["x-zen-source"] = "blazor-circuit",
        };

        return Build(key, "POST", "/beacons", "", pageUrl, "text/plain", null, null, headers, Encoding.UTF8.GetBytes(body));
    }

    private BeaconEntry Build(
        string key,
        string method,
        string path,
        string query,
        string? referer,
        string? contentType,
        string? client,
        string? userAgent,
        Dictionary<string, string> headers,
        ReadOnlySpan<byte> body)
    {
        var receivedAt = time.GetUtcNow();
        var isText = Utf8.IsValid(body) && !body.Contains((byte)0);
        var text = isText ? Encoding.UTF8.GetString(body) : Convert.ToBase64String(body);

        var cleanPath = Clean(path, MaxPathLength);
        var cleanQuery = Clean(query, MaxQueryLength);
        var cleanReferer = Clean(referer, MaxRefererLength);
        var cleanContentType = NullIfEmpty(Clean(contentType, MaxContentTypeLength));
        var cleanUserAgent = NullIfEmpty(Clean(userAgent, MaxUserAgentLength));

        var size = RowOverhead
            + Encoding.UTF8.GetByteCount(text)
            + key.Length
            + cleanPath.Length
            + cleanQuery.Length
            + cleanReferer.Length
            + (cleanContentType?.Length ?? 0)
            + (cleanUserAgent?.Length ?? 0)
            + headers.Sum(h => h.Key.Length + h.Value.Length + 6);

        return new BeaconEntry(
            Guid.CreateVersion7(receivedAt),
            key,
            receivedAt,
            Clean(method, 16),
            cleanPath,
            cleanQuery,
            cleanReferer,
            cleanContentType,
            client,
            cleanUserAgent,
            headers,
            text,
            isText ? Utf8Encoding : Base64Encoding,
            size);
    }

    private Dictionary<string, string> CaptureHeaders(IHeaderDictionary source)
    {
        var headers = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        foreach (var (name, values) in source)
        {
            if (headers.Count >= _options.MaxHeaders)
            {
                break;
            }

            var lower = Clean(name, 128).ToLowerInvariant();
            if (lower.Length == 0)
            {
                continue;
            }

            headers[lower] = RedactedHeaders.Contains(lower)
                ? Redacted
                : Clean(values.ToString(), _options.MaxHeaderValueLength);
        }

        return headers;
    }

    public static string? Anonymize(IPAddress? address)
    {
        if (address is null)
        {
            return null;
        }

        if (address.IsIPv4MappedToIPv6)
        {
            address = address.MapToIPv4();
        }

        var bytes = address.GetAddressBytes();
        if (address.AddressFamily == AddressFamily.InterNetwork)
        {
            bytes[3] = 0;
            return $"{new IPAddress(bytes)}/24";
        }

        Array.Clear(bytes, 6, bytes.Length - 6);
        return $"{new IPAddress(bytes)}/48";
    }

    public static string Clean(string? value, int maxLength)
    {
        if (string.IsNullOrEmpty(value))
        {
            return "";
        }

        var builder = new StringBuilder(Math.Min(value.Length, maxLength));
        foreach (var c in value)
        {
            if (builder.Length >= maxLength)
            {
                break;
            }

            if (!char.IsControl(c))
            {
                builder.Append(c);
            }
        }

        if (builder.Length > 0 && char.IsHighSurrogate(builder[^1]))
        {
            builder.Length--;
        }

        return builder.ToString();
    }

    private static string? NullIfEmpty(string value) => value.Length == 0 ? null : value;
}

using Microsoft.Extensions.Options;

namespace Zen.Blazor.Beacons;

public sealed class BeaconJanitor(
    IBeaconStore store,
    IBeaconMaintenance maintenance,
    DatabaseCircuit circuit,
    IOptions<BeaconOptions> options,
    TimeProvider time,
    ILogger<BeaconJanitor> logger) : BackgroundService
{
    private const double TargetRatio = 0.8;
    private const double RecoverRatio = 0.6;
    private const long MinBudgetBytes = 1024 * 1024;
    private static readonly TimeSpan InitialDelay = TimeSpan.FromSeconds(15);

    private readonly BeaconOptions _options = options.Value;
    private long _budget = options.Value.StorageBudgetBytes;
    private long _lastDatabaseBytes;
    private PruneReport? _lastReport;

    public PruneReport? LastReport => Volatile.Read(ref _lastReport);

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        if (!store.IsConfigured)
        {
            return;
        }

        try
        {
            await Task.Delay(InitialDelay, time, stoppingToken);
            using var timer = new PeriodicTimer(_options.PruneInterval, time);
            do
            {
                if (!circuit.CanAttempt)
                {
                    continue;
                }

                try
                {
                    await RunAsync(stoppingToken);
                }
                catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
                {
                    throw;
                }
                catch (Exception ex)
                {
                    circuit.Failed(ex);
                }
            }
            while (await timer.WaitForNextTickAsync(stoppingToken));
        }
        catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
        {
        }
    }

    private async Task RunAsync(CancellationToken cancellationToken)
    {
        var cutoff = time.GetUtcNow().AddDays(-_options.RetentionDays);
        var result = await PruneAsync(cutoff, cancellationToken);

        if (result.DatabaseBytes > _options.DatabaseLimitBytes)
        {
            var growing = result.DatabaseBytes > _lastDatabaseBytes;
            if (growing)
            {
                _budget = Math.Max(MinBudgetBytes, _budget / 2);
                logger.LogWarning(
                    "Database at {Size} bytes exceeds {Limit}; budget lowered to {Budget}",
                    result.DatabaseBytes, _options.DatabaseLimitBytes, _budget);
                result = await PruneAsync(cutoff, cancellationToken);
            }

            await VacuumAsync(cancellationToken);
        }
        else if (result.DatabaseBytes < _options.DatabaseLimitBytes * RecoverRatio && _budget < _options.StorageBudgetBytes)
        {
            _budget = Math.Min(_options.StorageBudgetBytes, _budget + _budget / 4);
        }

        _lastDatabaseBytes = result.DatabaseBytes;
        circuit.Succeeded();
        Volatile.Write(ref _lastReport, new PruneReport(time.GetUtcNow(), result.Removed, result.StoredBytes, result.DatabaseBytes, _budget));
    }

    private async Task VacuumAsync(CancellationToken cancellationToken)
    {
        try
        {
            await maintenance.VacuumAsync(cancellationToken);
        }
        catch (Exception ex) when (!cancellationToken.IsCancellationRequested)
        {
            logger.LogWarning(ex, "Vacuum failed");
        }
    }

    private async Task<PruneResult> PruneAsync(DateTimeOffset cutoff, CancellationToken cancellationToken)
    {
        var result = await maintenance.PruneAsync(cutoff, _budget, (long)(_budget * TargetRatio), cancellationToken);
        if (result.Removed > 0)
        {
            logger.LogInformation("Pruned {Count} beacons; {Stored} bytes stored, database {Database} bytes", result.Removed, result.StoredBytes, result.DatabaseBytes);
        }

        return result;
    }
}

        var (stored, ok) = await TryStoreAsync(s => s.SummarizeAsync(key, referer, since, cancellationToken), cancellationToken);

        var memory = BeaconTotals.From(
            Memory(ok).Where(e => e.Key == key && (referer is null || e.Referer == referer)),
            since);

        var totals = (stored ?? BeaconTotals.Empty).Combine(memory);
        return new BeaconSummary(key, referer, totals.Total, totals.Bytes, totals.FirstAt, totals.LastAt, totals.Referers, totals.Hourly, IsDegraded(ok));
    }

    public async Task<BeaconKeys> TopKeysAsync(int limit, CancellationToken cancellationToken)
    {
        limit = Math.Clamp(limit, 1, MaxTopKeys);
        var (stored, ok) = await TryStoreAsync(s => s.TopKeysAsync(limit, cancellationToken), cancellationToken);
        var keys = (stored ?? [])
            .Concat(Memory(ok).GroupBy(e => e.Key, StringComparer.Ordinal).Select(g => new KeyCount(g.Key, g.LongCount())))
            .GroupBy(k => k.Key, StringComparer.Ordinal)
            .Select(g => new KeyCount(g.Key, g.Sum(k => k.Count)))
            .OrderByDescending(k => k.Count)
            .ThenBy(k => k.Key, StringComparer.Ordinal)
            .Take(limit)
            .ToList();

        return new BeaconKeys(keys, IsDegraded(ok));
    }

    public StorageStatus Status()
    {
        var stats = buffer.Stats;
        var state = circuit.State;
        var report = janitor.LastReport;
        var database = !store.IsConfigured ? "memory-only"
            : state.Failures > 0 ? "unavailable"
            : state.LastSuccessAt is null ? "connecting"
            : "connected";

        return new StorageStatus(
            database,
            state.LastError,
            state.RetryAt,
            state.LastSuccessAt,
            stats.Entries,
            stats.Bytes,
            stats.Unsaved,
            stats.Dropped,
            report?.StoredBytes,
            report?.DatabaseBytes,
            report?.BudgetBytes ?? _options.StorageBudgetBytes,
            _options.DatabaseLimitBytes,
            _options.RetentionDays,
            _options.MaxBodyBytes,
            report?.At);
    }

    public async Task<BeaconOverview> GetOverviewAsync(IReadOnlyCollection<string> countKeys, bool fresh, CancellationToken cancellationToken)
    {
        var signature = string.Join('\n', countKeys);
        var requestedAt = time.GetUtcNow();

        if (!fresh && Volatile.Read(ref _overview) is { } cached && cached.Signature == signature && requestedAt - cached.Value.At < OverviewTtl)
        {
            return cached.Value;
        }

        await _overviewGate.WaitAsync(cancellationToken);
        try
        {
            if (_overview is { } current && current.Signature == signature
                && (fresh ? current.Value.At >= requestedAt : time.GetUtcNow() - current.Value.At < OverviewTtl))
            {
                return current.Value;
            }

            var overview = await BuildOverviewAsync(countKeys, cancellationToken);
            Volatile.Write(ref _overview, new OverviewCache(signature, overview));
            return overview;
        }
        finally
        {
            _overviewGate.Release();
        }
    }

    private async Task<BeaconOverview> BuildOverviewAsync(IReadOnlyCollection<string> countKeys, CancellationToken cancellationToken)
    {
        var at = time.GetUtcNow();
        var (stored, ok) = await TryStoreAsync(s => s.CountAsync(countKeys, cancellationToken), cancellationToken);
        var memory = Memory(ok).Where(e => countKeys.Contains(e.Key)).ToList();
        var counts = countKeys.Distinct(StringComparer.Ordinal).ToDictionary(
            k => k,
            k => (stored?.GetValueOrDefault(k) ?? 0) + memory.Count(e => e.Key == k),
            StringComparer.Ordinal);

        var recent = await ListAsync(null, null, null, 15, cancellationToken);
        var keys = await TopKeysAsync(10, cancellationToken);
        var degraded = IsDegraded(ok) || recent.Degraded || keys.Degraded;

        return new BeaconOverview(counts, recent.Items, keys.Keys, Status(), degraded, at);
    }

    private IEnumerable<BeaconEntry> Memory(bool storeOk)
    {
        var snapshot = buffer.Snapshot();
        return storeOk
            ? snapshot.Where(b => !b.Persisted).Select(b => b.Entry)
            : snapshot.Select(b => b.Entry);
    }

    private bool IsDegraded(bool storeOk) => store.IsConfigured && !storeOk;

    private async Task<(T? Value, bool Ok)> TryStoreAsync<T>(Func<IBeaconStore, Task<T>> operation, CancellationToken cancellationToken)
    {
        if (!store.IsConfigured)
        {
            return (await operation(store), true);
        }

        if (!circuit.CanAttempt)
        {
            return (default, false);
        }

        try
        {
            var value = await operation(store);
            circuit.Succeeded();
            return (value, true);
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            throw;
        }
        catch (Exception ex)
        {
            circuit.Failed(ex);
            logger.LogDebug(ex, "Beacon read fell back to memory");
            return (default, false);
        }
    }

    private static BeaconItem ToItem(BeaconEntry entry) => new(
        entry.Id,
        entry.Key,
        entry.ReceivedAt,
        entry.Referer,
        entry.ContentType,
        entry.Size,
        entry.Body.Length > 160 ? entry.Body[..160] : entry.Body);

    private sealed record OverviewCache(string Signature, BeaconOverview Value);
}

using Microsoft.Extensions.Options;

namespace Zen.Blazor.Beacons;

public sealed class BeaconWriter(
    BeaconBuffer buffer,
    IBeaconStore store,
    DatabaseCircuit circuit,
    IOptions<BeaconOptions> options,
    ILogger<BeaconWriter> logger) : BackgroundService
{
    private static readonly TimeSpan IdleWait = TimeSpan.FromSeconds(5);
    private static readonly TimeSpan ShutdownFlush = TimeSpan.FromSeconds(5);

    private readonly int _batchSize = options.Value.WriteBatchSize;
    private long _reportedDropped;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        if (!store.IsConfigured)
        {
            logger.LogWarning("No PostgreSQL settings found; beacons are kept in memory only");
            return;
        }

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await FlushAsync(stoppingToken);
                ReportDrops();
                var wait = circuit.CanAttempt ? IdleWait : circuit.RetryIn;
                await buffer.WaitForWorkAsync(wait, stoppingToken);
                if (!circuit.CanAttempt)
                {
                    await Task.Delay(circuit.RetryIn, stoppingToken).ConfigureAwait(ConfigureAwaitOptions.SuppressThrowing);
                }
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
            {
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Beacon writer loop failed");
                await Task.Delay(IdleWait, stoppingToken).ConfigureAwait(ConfigureAwaitOptions.SuppressThrowing);
            }
        }

        using var timeout = new CancellationTokenSource(ShutdownFlush);
        try
        {
            await FlushAsync(timeout.Token);
        }
        catch (Exception ex)
        {
            logger.LogWarning(ex, "Final beacon flush incomplete; {Count} unsaved beacons lost", buffer.Stats.Unsaved);
        }
    }

    private async Task FlushAsync(CancellationToken cancellationToken)
    {
        while (circuit.CanAttempt)
        {
            var batch = buffer.TakeUnsaved(_batchSize);
            if (batch.Count == 0)
            {
                return;
            }

            try
            {
                await store.InsertAsync(batch, cancellationToken);
            }
            catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
            {
                throw;
            }
            catch (Exception ex)
            {
                circuit.Failed(ex);
                return;
            }

            buffer.MarkPersisted(batch);
            circuit.Succeeded();
        }
    }

    private void ReportDrops()
    {
        var dropped = buffer.Stats.Dropped;
        if (dropped > _reportedDropped)
        {
            logger.LogWarning("{Count} unsaved beacons dropped from a full memory buffer", dropped - _reportedDropped);
            _reportedDropped = dropped;
        }
    }
}

namespace Zen.Blazor.Beacons;

public readonly record struct CircuitState(int Failures, DateTimeOffset? RetryAt, DateTimeOffset? LastSuccessAt, string? LastError);

public sealed class DatabaseCircuit(TimeProvider time, ILogger<DatabaseCircuit> logger)
{
    private static readonly TimeSpan MinDelay = TimeSpan.FromSeconds(2);
    private static readonly TimeSpan MaxDelay = TimeSpan.FromSeconds(60);

    private readonly Lock _gate = new();
    private int _failures;
    private DateTimeOffset _retryAt = DateTimeOffset.MinValue;
    private DateTimeOffset? _lastSuccessAt;
    private string? _lastError;

    public bool CanAttempt
    {
        get
        {
            lock (_gate)
            {
                return time.GetUtcNow() >= _retryAt;
            }
        }
    }

    public TimeSpan RetryIn
    {
        get
        {
            lock (_gate)
            {
                var wait = _retryAt - time.GetUtcNow();
                return wait > TimeSpan.Zero ? wait : TimeSpan.Zero;
            }
        }
    }

    public CircuitState State
    {
        get
        {
            lock (_gate)
            {
                return new CircuitState(
                    _failures,
                    _failures > 0 ? _retryAt : null,
                    _lastSuccessAt,
                    _lastError);
            }
        }
    }

    public void Succeeded()
    {
        int previous;
        lock (_gate)
        {
            previous = _failures;
            _failures = 0;
            _retryAt = DateTimeOffset.MinValue;
            _lastSuccessAt = time.GetUtcNow();
            _lastError = null;
        }

        if (previous > 0)
        {
            logger.LogWarning("Database available again after {Failures} failed attempts", previous);
        }
    }

    public void Failed(Exception exception)
    {
        int failures;
        TimeSpan delay;
        lock (_gate)
        {
            failures = ++_failures;
            var factor = Math.Pow(2, Math.Min(failures - 1, 10));
            delay = TimeSpan.FromTicks(Math.Min(MaxDelay.Ticks, (long)(MinDelay.Ticks * factor)));
            _retryAt = time.GetUtcNow() + delay;
            _lastError = exception.GetType().Name;
        }

        if (failures == 1)
        {
            logger.LogWarning(exception, "Database unavailable; serving from memory, retrying in {Delay}", delay);
        }
        else
        {
            logger.LogDebug(exception, "Database still unavailable after {Failures} attempts; retrying in {Delay}", failures, delay);
        }
    }
}

    {

            return;
        }

        var rows = entries.Select(e => new

            FROM beacon {where}

        var totals = await grid.ReadSingleAsync<TotalsRow>();
        var referers = await grid.ReadAsync<RefererRow>();
        var hourly = await grid.ReadAsync<HourRow>();

        return new BeaconTotals(
            totals.Total,
            totals.Bytes,
            Utc(totals.FirstAt),
            Utc(totals.LastAt),
            [.. referers.Select(r => new RefererCount(r.Referer, r.Count, Utc(r.LastAt)))],
            [.. hourly.Select(h => new HourCount(Utc(h.Hour), h.Count))]);

    public async Task<PruneResult> PruneAsync(DateTimeOffset olderThan, long budgetBytes, long targetBytes, CancellationToken cancellationToken)
    {
        await using var connection = await OpenAsync(cancellationToken);

        var removed = (long)await connection.ExecuteAsync(new CommandDefinition(
            "DELETE FROM beacon WHERE id < @cutoff",
            new { cutoff = BeaconIds.Floor(olderThan) },
            cancellationToken: cancellationToken));

        var stored = await connection.ExecuteScalarAsync<long>(new CommandDefinition(
            "SELECT coalesce(sum(size_bytes), 0)::bigint FROM beacon",
            cancellationToken: cancellationToken));

        if (stored > budgetBytes)
        {
            const string cutoffSql = """
                SELECT id
                FROM (SELECT id, sum(size_bytes) OVER (ORDER BY id DESC) AS running FROM beacon) ranked
                WHERE running > @targetBytes
                ORDER BY id DESC
                LIMIT 1
                """;

            var cutoff = await connection.ExecuteScalarAsync<Guid?>(new CommandDefinition(
                cutoffSql, new { targetBytes }, cancellationToken: cancellationToken));

            if (cutoff is { } id)
            {
                var trimmed = await connection.ExecuteAsync(new CommandDefinition(
                    "DELETE FROM beacon WHERE id <= @id", new { id }, cancellationToken: cancellationToken));
                removed += trimmed;
                logger.LogInformation("Trimmed {Count} beacons to stay under {Budget} bytes", trimmed, budgetBytes);

                stored = await connection.ExecuteScalarAsync<long>(new CommandDefinition(
                    "SELECT coalesce(sum(size_bytes), 0)::bigint FROM beacon",
                    cancellationToken: cancellationToken));
            }
        }

        var databaseBytes = await connection.ExecuteScalarAsync<long>(new CommandDefinition(
            "SELECT pg_database_size(current_database())",
            cancellationToken: cancellationToken));

        return new PruneResult(removed, stored, databaseBytes);
    }

    private async Task<NpgsqlConnection> OpenAsync(CancellationToken cancellationToken)
    {
        var connection = await dataSource.OpenConnectionAsync(cancellationToken);
        if (_schemaReady)
        {
            return connection;
        }

        try
        {
            await EnsureSchemaAsync(connection, cancellationToken);
            return connection;
        }
        catch
        {
            await connection.DisposeAsync();
            throw;
        }
    }

    private async Task EnsureSchemaAsync(NpgsqlConnection connection, CancellationToken cancellationToken)
    {
        await _schemaGate.WaitAsync(cancellationToken);
        try
        {
            if (_schemaReady)
            {
                return;
            }

            await using var transaction = await connection.BeginTransactionAsync(cancellationToken);
            await connection.ExecuteAsync(new CommandDefinition(
                "SELECT pg_advisory_xact_lock(@id)", new { id = SchemaLockId }, transaction, cancellationToken: cancellationToken));
            await connection.ExecuteAsync(new CommandDefinition(SchemaSql, transaction: transaction, cancellationToken: cancellationToken));
            await transaction.CommitAsync(cancellationToken);
            _schemaReady = true;
            logger.LogInformation("Beacon schema ready");
        }
        finally
        {
            _schemaGate.Release();
        }
    }

    public static string? ConnectionString(IConfiguration configuration, int maxPoolSize)
    {
        var builder = new NpgsqlConnectionStringBuilder();

        if (!string.IsNullOrWhiteSpace(configuration[Host]))
        {
            builder.Host = configuration[Host];
            if (int.TryParse(configuration[Port], out var port))
            {
                builder.Port = port;
            }

            builder.Database = configuration[Database];
            builder.Username = configuration[User];
            builder.Password = configuration[Password];

        else if (Uri.TryCreate(configuration[Url], UriKind.Absolute, out var uri) && uri.Scheme is "postgres" or "postgresql")
        {
            builder.Host = uri.Host;
            if (uri.Port > 0)
            {
                builder.Port = uri.Port;
            }

            builder.Database = Uri.UnescapeDataString(uri.AbsolutePath.TrimStart('/'));
            var credentials = uri.UserInfo.Split(':', 2);
            builder.Username = Uri.UnescapeDataString(credentials[0]);
            if (credentials.Length > 1)
            {
                builder.Password = Uri.UnescapeDataString(credentials[1]);
            }
        }
        else
        {
            return null;
        }

        builder.Pooling = true;
        builder.MinPoolSize = 0;
        builder.MaxPoolSize = maxPoolSize;
        builder.ConnectionIdleLifetime = 60;
        builder.ConnectionPruningInterval = 10;
        builder.Timeout = 5;
        builder.CommandTimeout = 15;
        builder.KeepAlive = 30;
        builder.SslMode = SslMode.Prefer;

using System.Globalization;
using System.Text.RegularExpressions;

namespace Zen.Blazor.Services;

public sealed record ChatMessage(long Id, DateTimeOffset At, string Text);

public sealed partial class MessageBoard(ILogger<MessageBoard> logger)
{
    public const int MaxLength = 500;
    public const int Capacity = 50;

    private readonly Lock _gate = new();
    private ChatMessage[] _messages = [];
    private long _nextId;

    public event Action? Changed;

    public IReadOnlyList<ChatMessage> Messages => Volatile.Read(ref _messages);

    public bool TryPost(string? input)
    {
        var text = Normalize(input);
        if (text.Length == 0)
        {
            return false;
        }

        lock (_gate)
        {
            var current = _messages;
            var next = new ChatMessage[Math.Min(current.Length + 1, Capacity)];
            next[0] = new ChatMessage(++_nextId, DateTimeOffset.Now, text);
            Array.Copy(current, 0, next, 1, next.Length - 1);
            Volatile.Write(ref _messages, next);
        }

        Notify();
        return true;
    }

    public static string Normalize(string? input)
    {
        if (string.IsNullOrWhiteSpace(input))
        {
            return "";
        }

        var text = string.Concat(input.ReplaceLineEndings("\n").Where(c => c == '\n' || !char.IsControl(c)));
        text = ExtraBlankLines().Replace(text, "\n\n").Trim();

        if (text.Length > MaxLength)
        {
            var cut = 0;
            while (cut < text.Length)
            {
                var next = StringInfo.GetNextTextElementLength(text, cut);
                if (cut + next > MaxLength)
                {
                    break;
                }

                cut += next;
            }

            text = text[..cut].TrimEnd();
        }

        return text.Any(IsVisible) ? text : "";
    }

    private static bool IsVisible(char c) =>
        !char.IsWhiteSpace(c) && char.GetUnicodeCategory(c) != UnicodeCategory.Format;

    private void Notify()
    {
        if (Changed is not { } handlers)
        {
            return;
        }

        foreach (var handler in handlers.GetInvocationList().Cast<Action>())
        {
            try
            {
                handler();
            }
            catch (Exception ex)
            {
                logger.LogWarning(ex, "Message board subscriber failed");
            }
        }
    }

    [GeneratedRegex(@"\n{3,}")]
    private static partial Regex ExtraBlankLines();
}
